cmake_minimum_required(VERSION 3.20)
project(Gnarlaxx LANGUAGES C OBJC)
if(NOT APPLE)
    message(FATAL_ERROR "The first implementation currently targets macOS.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(GNARLAXX_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(GNARLAXX_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

add_subdirectory(libs/audio)
add_library(decoders STATIC implementations/c/decoders.c)
target_include_directories(decoders SYSTEM PUBLIC vendor/stb)
add_library(sokol STATIC implementations/c/platform.m)
target_include_directories(sokol SYSTEM PUBLIC vendor/sokol)
target_compile_definitions(sokol PUBLIC SOKOL_METAL)
target_compile_options(sokol PRIVATE -fobjc-arc)
target_link_libraries(sokol PUBLIC "-framework Cocoa" "-framework QuartzCore"
    "-framework Metal" "-framework MetalKit" "-framework AudioToolbox")

# Odin supplies main() and calls sapp_run; both languages use the same platform glue.
add_library(sokol_odin STATIC EXCLUDE_FROM_ALL implementations/c/platform.m)
target_include_directories(sokol_odin SYSTEM PUBLIC vendor/sokol)
target_compile_definitions(sokol_odin PUBLIC SOKOL_METAL PRIVATE SOKOL_NO_ENTRY)
target_compile_options(sokol_odin PRIVATE -fobjc-arc)
target_link_libraries(sokol_odin PUBLIC "-framework Cocoa" "-framework QuartzCore"
    "-framework Metal" "-framework MetalKit")

add_library(gnarlaxx_core STATIC implementations/c/game.c implementations/c/scores.c)
target_include_directories(gnarlaxx_core PUBLIC implementations/c)
target_compile_options(gnarlaxx_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(gnarlaxx implementations/c/main.c implementations/c/renderer.c implementations/c/presentation.c implementations/c/sound_bank.c)
target_link_libraries(gnarlaxx PRIVATE gnarlaxx_core sokol decoders gnarlaxx_audio)
target_compile_options(gnarlaxx PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(gnarlaxx PRIVATE GNARLAXX_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/assets")
include(CTest)
if(BUILD_TESTING)
    add_executable(gnarlaxx_tests tests/test_game.c)
    target_link_libraries(gnarlaxx_tests PRIVATE gnarlaxx_core)
    target_compile_options(gnarlaxx_tests PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(gnarlaxx_tests PRIVATE TEST_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/assets"
        TEST_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME c_core COMMAND gnarlaxx_tests)
    add_executable(gnarlaxx_memory_tests tests/test_game_memory.c)
    target_link_libraries(gnarlaxx_memory_tests PRIVATE gnarlaxx_core)
    target_compile_options(gnarlaxx_memory_tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME c_dynamic_memory COMMAND gnarlaxx_memory_tests)
    add_executable(gnarlaxx_audio_tests tests/test_audio.c implementations/c/sound_bank.c)
    target_include_directories(gnarlaxx_audio_tests PRIVATE implementations/c)
    target_link_libraries(gnarlaxx_audio_tests PRIVATE gnarlaxx_audio)
    target_compile_options(gnarlaxx_audio_tests PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(gnarlaxx_audio_tests PRIVATE TEST_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/assets"
        TEST_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME shared_audio COMMAND gnarlaxx_audio_tests)
endif()
